N_SAMPLES = 8000

parts = list(PART_COSTS.keys())
severities = ["minor", "moderate", "severe"]

severity_weights = [0.25, 0.35, 0.40]

records = []

for _ in range(N_SAMPLES):

    part = np.random.choice(parts)

    severity = np.random.choice(
        severities,
        p=severity_weights
    )

    base = PART_COSTS[part][severity]

    # Damage area ratio
    if severity == "minor":
        area_ratio = np.random.uniform(0.02, 0.20)

    elif severity == "moderate":
        area_ratio = np.random.uniform(0.15, 0.50)

    else:
        area_ratio = np.random.uniform(0.40, 0.90)

    # Model confidence values
    yolo_confidence = np.random.uniform(0.45, 0.98)

    severity_confidence = np.random.uniform(0.40, 0.98)

    # Add realistic variation to labor
    labor_hours = base["labor_hours"] * np.random.uniform(
        0.80, 1.25
    )

    # Parts cost variation
    parts_cost = base["parts"] * np.random.uniform(
        0.80, 1.25
    )

    # Paint variation
    paint_cost = base["paint"] * np.random.uniform(
        0.80, 1.20
    )

    # Labor cost
    labor_cost = labor_hours * LABOR_RATE

    # Base repair cost
    total_cost = (
        parts_cost
        + labor_cost
        + paint_cost
    )

    # Larger damage generally increases repair cost
    area_factor = 0.75 + area_ratio * 0.50

    total_cost *= area_factor

    # Small real-world price variation
    noise = np.random.normal(
        loc=1.0,
        scale=0.07
    )

    total_cost *= noise

    total_cost = max(total_cost, 500)

    records.append({
        "part": part,
        "severity": severity,
        "damage_area_ratio": round(area_ratio, 4),
        "yolo_confidence": round(yolo_confidence, 4),
        "severity_confidence": round(severity_confidence, 4),
        "labor_hours": round(labor_hours, 2),
        "parts_cost": round(parts_cost, 2),
        "paint_cost": round(paint_cost, 2),
        "labor_cost": round(labor_cost, 2),
        "total_repair_cost": round(total_cost, 2)
    })

import cv2

def predict_severity(crop):
    """
    Predict severity from an automatically generated YOLO crop.
    """

    # Convert BGR → RGB
    crop_rgb = cv2.cvtColor(crop, cv2.COLOR_BGR2RGB)

    # Resize exactly as used during EfficientNet training
    crop_resized = cv2.resize(
        crop_rgb,
        (224, 224)
    )

    # EfficientNet model was receiving pixel values in 0–255 range
    image_array = crop_resized.astype(np.float32)

    # Add batch dimension
    image_array = np.expand_dims(
        image_array,
        axis=0
    )

    # Predict
    probabilities = severity_model.predict(
        image_array,
        verbose=0
    )[0]

    predicted_id = int(
        np.argmax(probabilities)
    )

    severity = SEVERITY_CLASSES[predicted_id]

    severity_confidence = float(
        probabilities[predicted_id]
    )

    return severity, severity_confidence, probabilities


def calculate_damage_area_ratio(
    bbox,
    image_width,
    image_height
):
    """
    Calculate bounding-box area / complete image area.
    """

    x1, y1, x2, y2 = bbox

    box_width = max(0, x2 - x1)
    box_height = max(0, y2 - y1)

    damage_area = box_width * box_height
    image_area = image_width * image_height

    if image_area == 0:
        return 0.0

    return float(
        damage_area / image_area
    )

    part,

        input_df
    )[0]

    # Cost cannot logically be negative
    predicted_cost = max(
        0,
        float(predicted_cost)
    )

    return predicted_cost
