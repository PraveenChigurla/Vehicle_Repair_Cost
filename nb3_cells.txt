def analyze_vehicle(
    image_path,
    yolo_model,
    severity_model,
    yolo_conf=0.25,
    save_path=None
):
    """
    Complete Vehicle Damage Assessment pipeline.

    Input:
        Original vehicle image

    Pipeline:
        YOLO detection
            ↓
        Automatic bounding-box crop
            ↓
        EfficientNet V4 severity classification
            ↓
        Annotated original image

    Returns:
        output_image
        detections
    """

    # =========================================================
    # 1. Load image
    # =========================================================

    image = cv2.imread(image_path)

    if image is None:
        raise FileNotFoundError(
            f"Could not read image:\n{image_path}"
        )

    output_image = image.copy()

    # =========================================================
    # 2. YOLO detection
    # =========================================================

    yolo_results = yolo_model.predict(
        source=image_path,
        conf=yolo_conf,
        verbose=False
    )

    result = yolo_results[0]

    detections = []

    # =========================================================
    # 3. Process every detected damaged part
    # =========================================================

    for i in range(len(result.boxes)):

        box = result.boxes[i]

        # -----------------------------------------------------
        # Bounding box
        # -----------------------------------------------------

        x1, y1, x2, y2 = (
            box.xyxy[0]
            .cpu()
            .numpy()
            .astype(int)
        )

        # -----------------------------------------------------
        # YOLO information
        # -----------------------------------------------------

        yolo_confidence = float(
            box.conf[0].cpu().numpy()
        )

        class_id = int(
            box.cls[0].cpu().numpy()
        )

        part_name = yolo_model.names[class_id]

        # -----------------------------------------------------
        # Keep coordinates inside image
        # -----------------------------------------------------

        h, w = image.shape[:2]

        x1 = max(0, min(int(x1), w - 1))
        y1 = max(0, min(int(y1), h - 1))
        x2 = max(0, min(int(x2), w))
        y2 = max(0, min(int(y2), h))

        if x2 <= x1 or y2 <= y1:
            continue

        # =====================================================
        # 4. AUTOMATIC CROP
        # =====================================================

        crop = image[y1:y2, x1:x2]

        if crop.size == 0:
            continue

        # =====================================================
        # 5. Prepare crop for EfficientNet
        # =====================================================

        crop_rgb = cv2.cvtColor(
            crop,
            cv2.COLOR_BGR2RGB
        )

        crop_resized = cv2.resize(
            crop_rgb,
            (224, 224)
        )

        crop_input = (
            np.expand_dims(
                crop_resized.astype(np.float32),
                axis=0
            )
        )

        # =====================================================
        # 6. Severity prediction
        # =====================================================

        probabilities = severity_model.predict(
            crop_input,
            verbose=0
        )[0]

        severity_id = int(
            np.argmax(probabilities)
        )

        severity = SEVERITY_CLASSES[
            severity_id
        ]

        severity_confidence = float(
            probabilities[severity_id]
        )

        # =====================================================
        # 7. Store detection
        # =====================================================

        detection = {
            "part": part_name,
            "yolo_confidence": yolo_confidence,
            "severity": severity,
            "severity_confidence": severity_confidence,
            "bbox": (
                int(x1),
                int(y1),
                int(x2),
                int(y2)
            )
        }

        detections.append(detection)

        # =====================================================
        # 8. Draw result
        # =====================================================

        output_image = draw_detection(
            output_image,
            detection["bbox"],
            detection["part"],
            detection["severity"],
            detection["severity_confidence"],
            detection["yolo_confidence"]
        )

    # =========================================================
    # 9. Save result
    # =========================================================

    if save_path is not None:

        os.makedirs(
            os.path.dirname(save_path),
            exist_ok=True
        )

        cv2.imwrite(
            save_path,
            output_image
        )

        print(
            f"\n✅ Annotated image saved to:\n"
            f"{save_path}"
        )

    return output_image, detections

def predict_repair_cost(
    part,
    severity,
    damage_area_ratio,
    yolo_confidence,
    severity_confidence
):
    """
    Predict repair cost using the trained XGBoost model.

    cost_model is already a Pipeline containing
    the preprocessing step, so raw features are passed directly.
    """

    input_df = pd.DataFrame([{
        "part": part,
        "severity": severity,
        "damage_area_ratio": damage_area_ratio,
        "yolo_confidence": yolo_confidence,
        "severity_confidence": severity_confidence
    }])

    # cost_model already contains the preprocessor.
    # Therefore, DO NOT call cost_preprocessor.transform()
    # separately here.
    predicted_cost = cost_model.predict(
        input_df
    )[0]

    # Cost cannot logically be negative
    predicted_cost = max(
        0,
        float(predicted_cost)
    )

    return predicted_cost
